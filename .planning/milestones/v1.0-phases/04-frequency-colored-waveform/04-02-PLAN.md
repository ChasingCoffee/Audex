// Icon: 3 ascending rectangles for color mode, single bar for monochrome

    int iconPad = (int)(4 * dpiScale);

    int iconBottom = _toggleButtonRect.Bottom - iconPad;
    int iconWidth = _toggleButtonRect.Width - iconPad * 2;
    int iconHeight = _toggleButtonRect.Height - iconPad * 2;
    int barW = Math.Max(1, iconWidth / 5);

    if (isColorMode)

        int h2 = iconHeight * 7 / 10;
